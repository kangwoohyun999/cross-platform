/* ************************************************************************************
*                                 인라인 형태 A X B 계산기(useSate 적용)
************************************************************************************* */
import React, { useState } from 'react';
import { View, Text, TextInput, Button } from 'react-native';

export default function App() {
  const [a, setA] = useState('');     // 첫 번째 숫자 (문자열)
  const [b, setB] = useState('');     // 두 번째 숫자 (문자열)
  const [result, setResult] = useState(null); // 계산 결과

  const multiply = () => {
    const numA = parseFloat(a);
    const numB = parseFloat(b);

    if (isNaN(numA) || isNaN(numB)) {
      setResult('숫자를 입력하세요!');
    } else {
      setResult(numA * numB);
    }
  };

  return (
    <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center' }}>
      <Text style={{ fontSize: 20, marginBottom: 10 }}>A × B 계산기</Text>
      <TextInput
        style={{
          borderWidth: 1,
          borderColor: 'gray',
          width: 200,
          marginBottom: 10,
          textAlign: 'center',
          fontSize: 18,
          borderRadius: 5,
        }}
        keyboardType="numeric"
        placeholder="A 입력"
        value={a}
        onChangeText={setA}
      />

      <TextInput
        style={{
          borderWidth: 1,
          borderColor: 'gray',
          width: 200,
          marginBottom: 10,
          textAlign: 'center',
          fontSize: 18,
          borderRadius: 5,
        }}
        keyboardType="numeric"
        placeholder="B 입력"
        value={b}
        onChangeText={setB}
      />

      <Button title="A × B 계산" onPress={multiply} />

      {result !== null && (
        <Text style={{ fontSize: 22, marginTop: 20 }}>결과: {result}</Text>
      )}
    </View>
  );
}
