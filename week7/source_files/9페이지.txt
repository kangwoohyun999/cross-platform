/* ************************************************************************************
 *                         useRef와 일반변수의 차이
 ************************************************************************************ */
import React, { useRef, useState } from 'react';
import { View, Button, Text } from 'react-native';

export default function Example() {
  const [render, setRender] = useState(false);
  const countRef = useRef(0);
  let countVar = 5;

  const handleClick = () => {
    countRef.current += 1;
    countVar += 1;
    console.log('현재 count:', countRef.current, ": ", countVar);
  };
  const doRender = () =>{
    setRender(!render);
  }
  return (
    <View
      style={{
        flex: 1,                     // 전체 화면 차지
        justifyContent: 'center',    // 세로 중앙 정렬
        alignItems: 'stretch',        // 가로 중앙 정렬
        padding:30,
      }}>
      <Text style={{fontSize:30, marginTop:10, marginBottom:30}}>결과값 CountRef:{countRef.current}, {"\n"}결과값 CountVar: {countVar} </Text>
      <View style={{ marginBottom: 12 }}>
        <Button title="클릭" onPress={handleClick} />
      </View>
      <View>
        <Button title="리랜더" onPress={doRender} />
      </View>
    </View>
  );
}
